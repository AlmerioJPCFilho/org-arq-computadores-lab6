lw x1,0(x0)       // x1 = 0x400 (1024 = endereços base MMIO)
lw x13,8(x0)      // x13 = 0x1FF (511 = 9 LEDG ligados)
sub x4,x1,x13     // x4 = 0x400 - 0x1FF = 0x201
and x4,x4,x13     // x4 = 0x201 AND 0x1FF = 1 // gera o resultado 1 
add x8,x4,x4      // x8 = 2  (KEY[1])
add x7,x8,x8      // x7 = 4  (KEY[2])
add x6,x7,x7      // x6 = 8  (KEY[3])
add x5,x6,x7      // x5 = 12
add x5,x5,x8      // x5 = 14
add x5,x5,x4      // x5 = 15 = 0xF (4 botoes soltos)
add x2,x0,x0      // x2 = contador = 0
add x3,x5,x0      // x3 = estado anterior dos botoes = 0xF (botos soltos)
lw x9,4(x1)       // loop: x9 = KEY atual (0x404), 0 = pressionado, 1 = solto
sub x10,x5,x9     // x10 = botoes pressionados agora (1 = pressionado, 0 = solto)
and x11,x3,x10    // x11 = botoes que ACABARAM de ser pressionados
add x3,x9,x0      // anterior = atual
and x12,x11,x6    // KEY[3] apertado agora?
beq x12,x0,8      // nao apertou: pula o acrescimo
add x2,x2,x4      // contador++
and x12,x11,x7    // KEY[2] apertado agora?
beq x12,x0,8      // nao apertou: pula o decremento
sub x2,x2,x4      // contador--
and x12,x11,x8    // KEY[1] apertado agora?
beq x12,x0,8      // nao apertou: pula o zeramento
add x2,x0,x0      // contador = 0
sw x2,12(x1)      // LEDG <= contador (0x40C)
beq x0,x0,-56     // volta para LOOP
